4.1 JOINs - Unir Tablas

Setup - Tablas de Ejemplo

-- Tabla clientes
CREATE TABLE clientes (
    cliente_id SERIAL PRIMARY KEY,
    nombre VARCHAR(100),
    email VARCHAR(255),
    ciudad VARCHAR(50)
);

-- Tabla pedidos
CREATE TABLE pedidos (
    pedido_id SERIAL PRIMARY KEY,
    cliente_id INTEGER REFERENCES clientes(cliente_id),
    producto VARCHAR(100),
    monto DECIMAL(10,2),
    fecha DATE
);

-- Datos de prueba
INSERT INTO clientes VALUES
(1, 'Juan Pérez', 'juan@email.com', 'Madrid'),
(2, 'María García', 'maria@email.com', 'Barcelona'),
(3, 'Carlos López', 'carlos@email.com', 'Madrid'),
(4, 'Ana Martínez', 'ana@email.com', 'Valencia');

INSERT INTO pedidos VALUES
(1, 1, 'Laptop', 1200, '2024-01-15'),
(2, 1, 'Mouse', 25, '2024-01-16'),
(3, 2, 'Teclado', 75, '2024-01-17'),
(4, 3, 'Monitor', 300, '2024-01-18');

INNER JOIN

-- Devuelve solo filas que tienen coincidencia en ambas tablas
SELECT 
    c.nombre AS cliente,
    c.email,
    p.producto,
    p.monto,
    p.fecha
FROM clientes c
INNER JOIN pedidos p ON c.cliente_id = p.cliente_id
ORDER BY p.fecha;

-- Resultado: Solo clientes con pedidos (Ana no aparece)

LEFT JOIN (LEFT OUTER JOIN)

-- Devuelve TODOS los clientes, incluso sin pedidos
SELECT 
    c.nombre AS cliente,
    c.email,
    p.producto,
    p.monto
FROM clientes c
LEFT JOIN pedidos p ON c.cliente_id = p.cliente_id
ORDER BY c.nombre;

-- Resultado: Todos los clientes (Ana aparece con NULL en pedidos)

RIGHT JOIN

-- Devuelve TODOS los pedidos, incluso sin cliente
SELECT 
    c.nombre AS cliente,
    p.producto,
    p.monto
FROM clientes c
RIGHT JOIN pedidos p ON c.cliente_id = p.cliente_id;

FULL OUTER JOIN

-- Devuelve TODAS las filas de ambas tablas
SELECT 
    c.nombre AS cliente,
    p.producto,
    p.monto
FROM clientes c
FULL OUTER JOIN pedidos p ON c.cliente_id = p.cliente_id;

CROSS JOIN

-- Producto cartesiano: todas las combinaciones posibles
SELECT c.nombre, p.producto
FROM clientes c
CROSS JOIN pedidos p;
-- Resultado: 4 clientes × 4 pedidos = 16 filas

SELF JOIN

-- Unir una tabla consigo misma
-- Ejemplo: Encontrar clientes de la misma ciudad
SELECT 
    c1.nombre AS cliente1,
    c2.nombre AS cliente2,
    c1.ciudad
FROM clientes c1
INNER JOIN clientes c2 ON c1.ciudad = c2.ciudad 
    AND c1.cliente_id < c2.cliente_id
ORDER BY c1.ciudad;

4.2 Subconsultas (Subqueries)

Subconsulta en WHERE

-- Clientes que han hecho pedidos mayores al promedio
SELECT nombre, email
FROM clientes
WHERE cliente_id IN (
    SELECT cliente_id 
    FROM pedidos 
    WHERE monto > (SELECT AVG(monto) FROM pedidos)
);

-- Clientes SIN pedidos
SELECT nombre, email
FROM clientes
WHERE cliente_id NOT IN (
    SELECT DISTINCT cliente_id FROM pedidos
);

Subconsulta en SELECT

-- Cada cliente con su total gastado
SELECT 
    c.nombre,
    c.email,
    (SELECT COALESCE(SUM(p.monto), 0) 
     FROM pedidos p 
     WHERE p.cliente_id = c.cliente_id) AS total_gastado
FROM clientes c;

Subconsulta en FROM

-- Usar resultado de consulta como tabla temporal
SELECT 
    ciudad,
    COUNT(*) AS num_clientes,
    AVG(total_gastado) AS promedio_gasto
FROM (
    SELECT 
        c.ciudad,
        c.cliente_id,
        COALESCE(SUM(p.monto), 0) AS total_gastado
    FROM clientes c
    LEFT JOIN pedidos p ON c.cliente_id = p.cliente_id
    GROUP BY c.ciudad, c.cliente_id
) AS subconsulta
GROUP BY ciudad;

Subconsulta Correlacionada

-- La subconsulta usa valores de la consulta externa
SELECT 
    c.nombre,
    c.email
FROM clientes c
WHERE EXISTS (
    SELECT 1 FROM pedidos p 
    WHERE p.cliente_id = c.cliente_id 
    AND p.monto > 500
);

4.3 Vistas (Views)

Crear Vista Simple

-- Vista que muestra pedidos con información del cliente
CREATE VIEW vista_pedidos_completos AS
SELECT 
    p.pedido_id,
    c.nombre AS cliente,
    c.email,
    c.ciudad,
    p.producto,
    p.monto,
    p.fecha
FROM pedidos p
INNER JOIN clientes c ON p.cliente_id = c.cliente_id;

-- Usar la vista
SELECT * FROM vista_pedidos_completos WHERE ciudad = 'Madrid';
SELECT cliente, SUM(monto) as total FROM vista_pedidos_completos GROUP BY cliente;

Vista con Agregaciones

CREATE VIEW resumen_clientes AS
SELECT 
    c.cliente_id,
    c.nombre,
    c.email,
    COUNT(p.pedido_id) AS num_pedidos,
    COALESCE(SUM(p.monto), 0) AS total_gastado,
    COALESCE(AVG(p.monto), 0) AS gasto_promedio,
    MAX(p.fecha) AS ultimo_pedido
FROM clientes c
LEFT JOIN pedidos p ON c.cliente_id = p.cliente_id
GROUP BY c.cliente_id, c.nombre, c.email;

-- Consultar la vista
SELECT * FROM resumen_clientes ORDER BY total_gastado DESC;

Vista Materializada

-- Se almacena físicamente y se actualiza manualmente
CREATE MATERIALIZED VIEW mv_resumen_ventas AS
SELECT 
    DATE_TRUNC('month', fecha) AS mes,
    COUNT(*) AS num_ventas,
    SUM(monto) AS total_ventas,
    AVG(monto) AS venta_promedio
FROM pedidos
GROUP BY DATE_TRUNC('month', fecha)
ORDER BY mes;

-- Consultar
SELECT * FROM mv_resumen_ventas;

-- Actualizar datos
REFRESH MATERIALIZED VIEW mv_resumen_ventas;

-- Actualizar concurrentemente (permite lecturas durante refresh)
REFRESH MATERIALIZED VIEW CONCURRENTLY mv_resumen_ventas;

Modificar y Eliminar Vistas

-- Reemplazar vista existente
CREATE OR REPLACE VIEW vista_pedidos_completos AS
SELECT p.*, c.nombre, c.email, c.ciudad
FROM pedidos p
JOIN clientes c ON p.cliente_id = c.cliente_id
WHERE p.monto > 10;

-- Eliminar vista
DROP VIEW IF EXISTS vista_pedidos_completos;
DROP MATERIALIZED VIEW IF EXISTS mv_resumen_ventas;

4.4 Funciones Almacenadas (Stored Procedures)

Crear Función Simple

-- Función que calcula el total de un cliente
CREATE OR REPLACE FUNCTION obtener_total_cliente(p_cliente_id INTEGER)
RETURNS DECIMAL AS $$
DECLARE
    total DECIMAL;
BEGIN
    SELECT COALESCE(SUM(monto), 0) INTO total
    FROM pedidos
    WHERE cliente_id = p_cliente_id;
    
    RETURN total;
END;
$$ LANGUAGE plpgsql;

-- Usar la función
SELECT 
    nombre, 
    email,
    obtener_total_cliente(cliente_id) AS total
FROM clientes;

Función con Múltiples Parámetros

CREATE OR REPLACE FUNCTION obtener_pedidos_rango(
    p_cliente_id INTEGER,
    p_fecha_inicio DATE,
    p_fecha_fin DATE
)
RETURNS TABLE (
    pedido_id INTEGER,
    producto VARCHAR,
    monto DECIMAL,
    fecha DATE
) AS $$
BEGIN
    RETURN QUERY
    SELECT p.pedido_id, p.producto, p.monto, p.fecha
    FROM pedidos p
    WHERE p.cliente_id = p_cliente_id
    AND p.fecha BETWEEN p_fecha_inicio AND p_fecha_fin;
END;
$$ LANGUAGE plpgsql;

-- Usar
SELECT * FROM obtener_pedidos_rango(1, '2024-01-01', '2024-12-31');

Función que Modifica Datos

CREATE OR REPLACE FUNCTION registrar_pedido(
    p_cliente_id INTEGER,
    p_producto VARCHAR,
    p_monto DECIMAL
)
RETURNS INTEGER AS $$
DECLARE
    nuevo_pedido_id INTEGER;
BEGIN
    INSERT INTO pedidos (cliente_id, producto, monto, fecha)
    VALUES (p_cliente_id, p_producto, p_monto, CURRENT_DATE)
    RETURNING pedido_id INTO nuevo_pedido_id;
    
    RETURN nuevo_pedido_id;
END;
$$ LANGUAGE plpgsql;

-- Usar
SELECT registrar_pedido(1, 'Tablet', 450);

Procedure (PostgreSQL 11+)

CREATE OR REPLACE PROCEDURE actualizar_precios(p_porcentaje DECIMAL)
LANGUAGE plpgsql
AS $$
BEGIN
    UPDATE pedidos
    SET monto = monto * (1 + p_porcentaje / 100);
    
    COMMIT;
END;
$$;

-- Llamar al procedure
CALL actualizar_precios(10);  -- Aumento del 10%

4.5 Triggers (Disparadores)

Trigger para Auditoría

-- Tabla de auditoría
CREATE TABLE auditoria_pedidos (
    auditoria_id SERIAL PRIMARY KEY,
    pedido_id INTEGER,
    accion VARCHAR(10),
    monto_anterior DECIMAL,
    monto_nuevo DECIMAL,
    usuario VARCHAR(100),
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Función trigger
CREATE OR REPLACE FUNCTION fn_auditoria_pedidos()
RETURNS TRIGGER AS $$
BEGIN
    IF TG_OP = 'INSERT' THEN
        INSERT INTO auditoria_pedidos (pedido_id, accion, monto_nuevo, usuario)
        VALUES (NEW.pedido_id, 'INSERT', NEW.monto, CURRENT_USER);
        RETURN NEW;
    ELSIF TG_OP = 'UPDATE' THEN
        INSERT INTO auditoria_pedidos (pedido_id, accion, monto_anterior, monto_nuevo, usuario)
        VALUES (NEW.pedido_id, 'UPDATE', OLD.monto, NEW.monto, CURRENT_USER);
        RETURN NEW;
    ELSIF TG_OP = 'DELETE' THEN
        INSERT INTO auditoria_pedidos (pedido_id, accion, monto_anterior, usuario)
        VALUES (OLD.pedido_id, 'DELETE', OLD.monto, CURRENT_USER);
        RETURN OLD;
    END IF;
    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

-- Crear trigger
CREATE TRIGGER trg_auditoria_pedidos
AFTER INSERT OR UPDATE OR DELETE ON pedidos
FOR EACH ROW EXECUTE FUNCTION fn_auditoria_pedidos();

Trigger para Validación

CREATE OR REPLACE FUNCTION fn_validar_monto_pedido()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.monto < 0 THEN
        RAISE EXCEPTION 'El monto del pedido no puede ser negativo';
    END IF;
    
    IF NEW.monto > 100000 THEN
        RAISE NOTICE 'Pedido de alto valor registrado';
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_validar_monto
BEFORE INSERT OR UPDATE ON pedidos
FOR EACH ROW EXECUTE FUNCTION fn_validar_monto_pedido();

Trigger para Actualización Automática

-- Tabla con timestamp de actualización
CREATE TABLE productos (
    producto_id SERIAL PRIMARY KEY,
    nombre VARCHAR(100),
    precio DECIMAL,
    actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Función para actualizar timestamp
CREATE OR REPLACE FUNCTION fn_actualizar_timestamp()
RETURNS TRIGGER AS $$
BEGIN
    NEW.actualizado_en = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Trigger
CREATE TRIGGER trg_actualizar_timestamp
BEFORE UPDATE ON productos
FOR EACH ROW EXECUTE FUNCTION fn_actualizar_timestamp();

4.7 Resumen del Tema

Puntos Clave

  • JOINs permiten combinar datos de múltiples tablas
  • Subconsultas pueden ir en WHERE, SELECT o FROM
  • Vistas simplifican consultas complejas y reutilizan código
  • Funciones almacenadas encapsulan lógica de negocio
  • Triggers ejecutan código automáticamente ante eventos
