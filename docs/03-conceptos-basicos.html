Tema 03: Conceptos Básicos de PostgreSQL

Aprende los fundamentos esenciales: bases de datos, tablas, tipos de datos, restricciones y operaciones CRUD.

3.1 Creación de Bases de Datos

Sintaxis Básica

-- Crear una base de datos simple
CREATE DATABASE nombre_base_datos;

-- Crear con opciones específicas
CREATE DATABASE mi_empresa
    WITH 
    ENCODING = 'UTF8'
    LC_COLLATE = 'es_ES.UTF-8'
    LC_CTYPE = 'es_ES.UTF-8'
    TEMPLATE = template0
    OWNER = postgres;

-- Ver bases de datos existentes
\l

-- Conectarse a una base de datos
\c nombre_base_datos;

-- Eliminar base de datos
DROP DATABASE IF EXISTS nombre_base_datos;

Opciones de CREATE DATABASE

OpciónDescripciónEjemplo
ENCODINGCodificación de caracteresUTF8, LATIN1
LC_COLLATEReglas de ordenamientoes_ES.UTF-8
LC_CTYPEClasificación de caractereses_ES.UTF-8
TEMPLATEBase de datos plantillatemplate0, template1
OWNERUsuario propietariopostgres, mi_usuario

3.2 Tipos de Datos en PostgreSQL

Tipos Numéricos

CREATE TABLE tipos_numericos (
    id SERIAL PRIMARY KEY,
    entero_pequeno SMALLINT,      -- -32768 a +32767
    entero_normal INTEGER,         -- -2B a +2B
    entero_grande BIGINT,          -- -9M a +9M
    decimal_exacto DECIMAL(10,2),  -- Exacto, 10 dígitos, 2 decimales
    punto_flotante REAL,           -- 6 dígitos precisión
    doble_precision DOUBLE PRECISION, -- 15 dígitos
    dinero MONEY                   -- Formato monetario
);

-- Ejemplos de inserción
INSERT INTO tipos_numericos VALUES
(1, 100, 50000, 1000000, 99999999.99, 3.14, 3.14159265358979, 1500.50);

Tipos de Caracteres

CREATE TABLE tipos_texto (
    id SERIAL PRIMARY KEY,
    texto_longitud_fija CHAR(50),      -- Siempre 50 caracteres (rellena con espacios)
    texto_longitud_variable VARCHAR(255), -- Hasta 255 caracteres
    texto_ilimitado TEXT,               -- Sin límite práctico
    nombre_nombre NAME                  -- Para nombres de objetos (63 bytes)
);

-- TEXT es el más recomendado en PostgreSQL
-- VARCHAR(n) y TEXT tienen el mismo rendimiento

Tipos de Fecha y Hora

CREATE TABLE tipos_fecha (
    id SERIAL PRIMARY KEY,
    solo_fecha DATE,                    -- YYYY-MM-DD
    solo_hora TIME,                     -- HH:MM:SS
    hora_con_zona TIME WITH TIME ZONE,  -- HH:MM:SS+TZ
    fecha_hora TIMESTAMP,               -- YYYY-MM-DD HH:MM:SS
    fecha_hora_zona TIMESTAMP WITH TIME ZONE, -- Con zona horaria
    intervalo INTERVAL                  -- Período de tiempo
);

-- Ejemplos
INSERT INTO tipos_fecha VALUES
(1, 
    '2024-01-15', 
    '14:30:00', 
    '14:30:00-05',
    '2024-01-15 14:30:00',
    '2024-01-15 14:30:00-05',
    INTERVAL '1 day 2 hours 30 minutes'
);

-- NOW() devuelve fecha y hora actual
SELECT NOW();
SELECT CURRENT_DATE;
SELECT CURRENT_TIME;

Tipos Booleanos

CREATE TABLE tipos_booleanos (
    id SERIAL PRIMARY KEY,
    activo BOOLEAN,
    verificado BOOL  -- BOOL es alias de BOOLEAN
);

-- Valores válidos: TRUE, FALSE, NULL
-- También acepta: 't', 'f', 'true', 'false', 'y', 'n', 'yes', 'no', '1', '0'
INSERT INTO tipos_booleanos VALUES
(1, TRUE, 't'),
(2, FALSE, 'no'),
(3, NULL, NULL);

Tipos JSON

CREATE TABLE tipos_json (
    id SERIAL PRIMARY KEY,
    json_basico JSON,      -- JSON como texto
    json_binario JSONB     -- JSON binario (más eficiente para consultas)
);

-- JSONB es recomendado para producción
INSERT INTO tipos_json VALUES
(1, 
    '{"nombre": "Juan", "edad": 30, "ciudad": "Madrid"}',
    '{"nombre": "Juan", "edad": 30, "ciudad": "Madrid"}'
);

-- Consultar datos JSON
SELECT json_basico->>'nombre' as nombre FROM tipos_json;
SELECT * FROM tipos_json WHERE jsonb_binario->>'edad' = '30';

Tipos Especiales

CREATE TABLE tipos_especiales (
    id SERIAL PRIMARY KEY,
    uuid UUID DEFAULT gen_random_uuid(),     -- Identificador único universal
    ip INET,                                  -- Dirección IPv4 o IPv6
    mac MACADDR,                              -- Dirección MAC
    rango_int INT4RANGE,                      -- Rango de enteros
    rango_fecha DATERANGE,                    -- Rango de fechas
    array_texto TEXT[],                       -- Array de texto
    array_numeros INTEGER[]                   -- Array de números
);

-- Ejemplo con arrays
INSERT INTO tipos_especiales (ip, array_texto, array_numeros) VALUES
('192.168.1.1', ARRAY['a', 'b', 'c'], ARRAY[1, 2, 3, 4, 5]);

-- Ejemplo con UUID
INSERT INTO tipos_especiales DEFAULT VALUES;
SELECT uuid FROM tipos_especiales;

3.3 Creación de Tablas

Sintaxis Completa

CREATE TABLE empleados (
    -- Columna con autoincremento
    empleado_id SERIAL PRIMARY KEY,
    
    -- Columnas con restricciones
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    telefono VARCHAR(20),
    
    -- Fecha con valor por defecto
    fecha_contratacion DATE DEFAULT CURRENT_DATE,
    
    -- Salario con validación
    salario DECIMAL(10,2) CHECK (salario >= 0),
    
    -- Departamento como clave foránea
    departamento_id INTEGER,
    
    -- Timestamp de creación
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    -- Clave foránea
    FOREIGN KEY (departamento_id) REFERENCES departamentos(departamento_id)
);

Crear Tabla desde Otra Tabla

-- Copiar estructura y datos
CREATE TABLE empleados_backup AS 
SELECT * FROM empleados;

-- Copiar solo estructura
CREATE TABLE empleados_estructura AS 
SELECT * FROM empleados WHERE 1 = 0;

-- Copiar columnas específicas
CREATE TABLE empleados_resumen AS
SELECT empleado_id, nombre, apellido, email FROM empleados;

3.4 Restricciones (Constraints)

Tipos de Restricciones

CREATE TABLE productos (
    producto_id SERIAL PRIMARY KEY,                    -- Clave primaria
    nombre VARCHAR(100) NOT NULL,                       -- No nulo
    sku VARCHAR(50) UNIQUE NOT NULL,                    -- Único
    precio DECIMAL(10,2) CHECK (precio > 0),           -- Validación
    categoria_id INTEGER REFERENCES categorias(id),    -- Clave foránea
    stock INTEGER DEFAULT 0 CHECK (stock >= 0),        -- Default + Check
    activo BOOLEAN DEFAULT TRUE,                        -- Valor por defecto
    CONSTRAINT precio_stock_check CHECK (precio > 0 AND stock >= 0)  -- Check múltiple
);

Agregar/Eliminar Restricciones

-- Agregar NOT NULL
ALTER TABLE productos ALTER COLUMN nombre SET NOT NULL;

-- Agregar UNIQUE
ALTER TABLE productos ADD CONSTRAINT unique_sku UNIQUE (sku);

-- Agregar CHECK
ALTER TABLE productos ADD CONSTRAINT precio_positivo CHECK (precio > 0);

-- Agregar FOREIGN KEY
ALTER TABLE ordenes 
ADD CONSTRAINT fk_producto 
FOREIGN KEY (producto_id) REFERENCES productos(producto_id);

-- Eliminar restricción
ALTER TABLE productos DROP CONSTRAINT precio_positivo;

-- Eliminar NOT NULL
ALTER TABLE productos ALTER COLUMN telefono DROP NOT NULL;

3.5 Operaciones CRUD

CREATE - Insertar Datos

-- Insertar una fila
INSERT INTO empleados (nombre, apellido, email, salario)
VALUES ('Juan', 'Pérez', 'juan@email.com', 50000);

-- Insertar múltiples filas
INSERT INTO empleados (nombre, apellido, email, salario) VALUES
('María', 'García', 'maria@email.com', 55000),
('Carlos', 'López', 'carlos@email.com', 48000),
('Ana', 'Martínez', 'ana@email.com', 52000);

-- Insertar y devolver valores
INSERT INTO empleados (nombre, apellido, email, salario)
VALUES ('Pedro', 'Sánchez', 'pedro@email.com', 60000)
RETURNING empleado_id, nombre, email;

-- Insertar desde otra tabla
INSERT INTO empleados_backup
SELECT * FROM empleados WHERE fecha_contratacion >= '2024-01-01';

READ - Consultar Datos

-- Consulta básica
SELECT * FROM empleados;

-- Columnas específicas
SELECT nombre, apellido, email FROM empleados;

-- Con alias
SELECT 
    nombre AS "Nombre Completo",
    email AS "Correo Electrónico",
    salario AS "Salario Anual"
FROM empleados;

-- DISTINCT - Valores únicos
SELECT DISTINCT departamento_id FROM empleados;

-- Con WHERE
SELECT * FROM empleados WHERE salario > 50000;

-- Múltiples condiciones
SELECT * FROM empleados 
WHERE salario > 50000 AND departamento_id = 1;

-- LIKE - Búsqueda de patrones
SELECT * FROM empleados WHERE nombre LIKE 'J%';  -- Empieza con J
SELECT * FROM empleados WHERE email LIKE '%@gmail.com';

-- ORDER BY
SELECT * FROM empleados ORDER BY salario DESC;

-- LIMIT y OFFSET
SELECT * FROM empleados ORDER BY salario DESC LIMIT 10 OFFSET 0;

-- Funciones de agregación
SELECT 
    COUNT(*) as total_empleados,
    AVG(salario) as salario_promedio,
    MAX(salario) as salario_maximo,
    MIN(salario) as salario_minimo,
    SUM(salario) as total_nomina
FROM empleados;

-- GROUP BY
SELECT departamento_id, COUNT(*) as cantidad, AVG(salario) as promedio
FROM empleados
GROUP BY departamento_id;

-- HAVING - Filtrar grupos
SELECT departamento_id, AVG(salario) as promedio
FROM empleados
GROUP BY departamento_id
HAVING AVG(salario) > 50000;

UPDATE - Actualizar Datos

-- Actualizar una fila
UPDATE empleados 
SET salario = 55000 
WHERE empleado_id = 1;

-- Actualizar múltiples columnas
UPDATE empleados 
SET salario = 60000, departamento_id = 2
WHERE empleado_id = 1;

-- Actualizar múltiples filas
UPDATE empleados 
SET salario = salario * 1.1  -- Aumento del 10%
WHERE departamento_id = 1;

-- Actualizar y devolver valores
UPDATE empleados 
SET salario = 65000 
WHERE empleado_id = 1
RETURNING *;

-- Actualizar con subconsulta
UPDATE empleados 
SET salario = (SELECT AVG(salario) FROM empleados)
WHERE empleado_id = 1;

DELETE - Eliminar Datos

-- Eliminar una fila
DELETE FROM empleados WHERE empleado_id = 1;

-- Eliminar múltiples filas
DELETE FROM empleados WHERE departamento_id = 5;

-- Eliminar y devolver valores
DELETE FROM empleados 
WHERE fecha_contratacion < '2020-01-01'
RETURNING *;

-- TRUNCATE - Eliminar todos los datos (más rápido)
TRUNCATE TABLE empleados;

-- TRUNCATE con reinicio de secuencia
TRUNCATE TABLE empleados RESTART IDENTITY;

3.6 Comandos psql Esenciales

-- Conectar a base de datos
\c nombre_base_datos

-- Listar bases de datos
\l

-- Listar tablas
\dt

-- Ver estructura de tabla
\d nombre_tabla

-- Ver todas las tablas con detalles
\d+

-- Listar esquemas
\dn

-- Listar usuarios
\du

-- Ver consultas recientes
\s

-- Mostrar historial de comandos
\s /ruta/archivo.sql

-- Ejecutar archivo SQL
\i /ruta/archivo.sql

-- Guardar salida en archivo
\o /ruta/salida.txt
SELECT * FROM empleados;
\o

-- Mostrar tiempo de ejecución
\timing

-- Ayuda sobre comandos SQL
\h SELECT
\h CREATE TABLE

-- Ayuda sobre comandos psql
\?

-- Limpiar pantalla
\! clear  -- Linux/Mac
\! cls    -- Windows

-- Salir
\q

3.8 Resumen del Tema

Puntos Clave

  • PostgreSQL soporta una amplia variedad de tipos de datos incluyendo JSONB y arrays
  • Las restricciones (constraints) aseguran la integridad de los datos
  • CRUD son las operaciones básicas: Create, Read, Update, Delete
  • psql tiene comandos especiales que comienzan con \
  • RETURNING permite obtener valores después de INSERT/UPDATE/DELETE
