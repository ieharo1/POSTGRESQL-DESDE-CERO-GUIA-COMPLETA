Tema 07: Proyecto Final - Sistema de Gestión Empresarial

Desarrolla una base de datos completa para una empresa con todos los conceptos aprendidos.

7.1 Descripción del Proyecto

Objetivo: Crear un sistema completo de gestión empresarial que incluya módulos de recursos humanos, inventario, ventas y reportes.

Requisitos del Sistema

  • Gestión de empleados y departamentos
  • Control de inventario de productos
  • Procesamiento de ventas y órdenes
  • Sistema de usuarios y permisos
  • Reportes y análisis de datos
  • Auditoría de todas las operaciones

7.2 Esquema de Base de Datos Completo

Módulo: Recursos Humanos

-- Departamentos
CREATE TABLE departamentos (
    departamento_id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion TEXT,
    presupuesto DECIMAL(12,2),
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Puestos
CREATE TABLE puestos (
    puesto_id SERIAL PRIMARY KEY,
    titulo VARCHAR(100) NOT NULL,
    descripcion TEXT,
    salario_minimo DECIMAL(10,2),
    salario_maximo DECIMAL(10,2),
    departamento_id INTEGER REFERENCES departamentos(departamento_id)
);

-- Empleados
CREATE TABLE empleados (
    empleado_id SERIAL PRIMARY KEY,
    numero_empleado VARCHAR(20) UNIQUE NOT NULL,
    nombre VARCHAR(100) NOT NULL,
    apellido_paterno VARCHAR(100) NOT NULL,
    apellido_materno VARCHAR(100),
    email VARCHAR(255) UNIQUE NOT NULL,
    telefono VARCHAR(20),
    fecha_nacimiento DATE,
    direccion TEXT,
    puesto_id INTEGER REFERENCES puestos(puesto_id),
    jefe_id INTEGER REFERENCES empleados(empleado_id),
    fecha_contratacion DATE DEFAULT CURRENT_DATE,
    salario DECIMAL(10,2) CHECK (salario >= 0),
    activo BOOLEAN DEFAULT true,
    eliminado_en TIMESTAMP,
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Asistencias
CREATE TABLE asistencias (
    asistencia_id SERIAL PRIMARY KEY,
    empleado_id INTEGER REFERENCES empleados(empleado_id),
    fecha DATE NOT NULL,
    hora_entrada TIME,
    hora_salida TIME,
    tipo VARCHAR(20) DEFAULT 'normal',  -- normal, falta, retardo, vacaciones
    justificante TEXT,
    UNIQUE(empleado_id, fecha)
);

Módulo: Inventario

-- Proveedores
CREATE TABLE proveedores (
    proveedor_id SERIAL PRIMARY KEY,
    nombre VARCHAR(200) NOT NULL,
    contacto VARCHAR(100),
    email VARCHAR(255),
    telefono VARCHAR(20),
    direccion TEXT,
    credito_limite DECIMAL(12,2),
    activo BOOLEAN DEFAULT true
);

-- Categorías de productos
CREATE TABLE categorias_producto (
    categoria_id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    padre_id INTEGER REFERENCES categorias_producto(categoria_id)
);

-- Productos
CREATE TABLE productos (
    producto_id SERIAL PRIMARY KEY,
    sku VARCHAR(50) UNIQUE NOT NULL,
    nombre VARCHAR(200) NOT NULL,
    descripcion TEXT,
    categoria_id INTEGER REFERENCES categorias_producto(categoria_id),
    proveedor_id INTEGER REFERENCES proveedores(proveedor_id),
    costo DECIMAL(10,2) CHECK (costo >= 0),
    precio_venta DECIMAL(10,2) CHECK (precio_venta >= 0),
    stock_actual INTEGER DEFAULT 0,
    stock_minimo INTEGER DEFAULT 10,
    stock_maximo INTEGER,
    ubicacion VARCHAR(50),
    activo BOOLEAN DEFAULT true,
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Movimientos de inventario
CREATE TABLE inventario_movimientos (
    movimiento_id SERIAL PRIMARY KEY,
    producto_id INTEGER REFERENCES productos(producto_id),
    tipo VARCHAR(20) NOT NULL,  -- entrada, salida, ajuste, transferencia
    cantidad INTEGER NOT NULL,
    costo_unitario DECIMAL(10,2),
    referencia VARCHAR(100),  -- número de orden, factura, etc.
    usuario_id INTEGER,
    notas TEXT,
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Módulo: Ventas

-- Clientes
CREATE TABLE clientes (
    cliente_id SERIAL PRIMARY KEY,
    tipo_cliente VARCHAR(20) DEFAULT 'individual',  -- individual, empresa
    nombre VARCHAR(200) NOT NULL,
    rfc VARCHAR(13),  -- Para empresas
    email VARCHAR(255),
    telefono VARCHAR(20),
    direccion_facturacion TEXT,
    direccion_envio TEXT,
    credito_limite DECIMAL(12,2),
    saldo_pendiente DECIMAL(12,2) DEFAULT 0,
    activo BOOLEAN DEFAULT true
);

-- Órdenes de venta
CREATE TABLE ventas (
    venta_id SERIAL PRIMARY KEY,
    folio VARCHAR(20) UNIQUE NOT NULL,
    cliente_id INTEGER REFERENCES clientes(cliente_id),
    vendedor_id INTEGER REFERENCES empleados(empleado_id),
    estado VARCHAR(30) DEFAULT 'pendiente',  -- pendiente, procesando, enviada, completada, cancelada
    subtotal DECIMAL(12,2),
    descuento DECIMAL(12,2) DEFAULT 0,
    impuesto DECIMAL(12,2),
    total DECIMAL(12,2),
    notas TEXT,
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Detalle de ventas
CREATE TABLE ventas_detalle (
    venta_id INTEGER REFERENCES ventas(venta_id),
    producto_id INTEGER REFERENCES productos(producto_id),
    cantidad INTEGER NOT NULL CHECK (cantidad > 0),
    precio_unitario DECIMAL(10,2) NOT NULL,
    descuento DECIMAL(10,2) DEFAULT 0,
    subtotal DECIMAL(12,2) NOT NULL,
    PRIMARY KEY (venta_id, producto_id)
);

-- Pagos
CREATE TABLE pagos (
    pago_id SERIAL PRIMARY KEY,
    venta_id INTEGER REFERENCES ventas(venta_id),
    monto DECIMAL(12,2) NOT NULL,
    metodo_pago VARCHAR(30),  -- efectivo, tarjeta, transferencia, cheque
    referencia VARCHAR(100),
    fecha_pago DATE DEFAULT CURRENT_DATE,
    notas TEXT
);

Módulo: Seguridad y Auditoría

-- Usuarios del sistema
CREATE TABLE usuarios_sistema (
    usuario_id SERIAL PRIMARY KEY,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    empleado_id INTEGER REFERENCES empleados(empleado_id),
    activo BOOLEAN DEFAULT true,
    ultimo_acceso TIMESTAMP,
    intentos_fallidos INTEGER DEFAULT 0,
    bloqueado_hasta TIMESTAMP,
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Roles
CREATE TABLE roles (
    rol_id SERIAL PRIMARY KEY,
    nombre VARCHAR(50) UNIQUE NOT NULL,
    descripcion TEXT,
    es_sistema BOOLEAN DEFAULT false  -- Roles del sistema no se pueden eliminar
);

-- Usuarios-Roles
CREATE TABLE usuarios_roles (
    usuario_id INTEGER REFERENCES usuarios_sistema(usuario_id),
    rol_id INTEGER REFERENCES roles(rol_id),
    PRIMARY KEY (usuario_id, rol_id)
);

-- Permisos
CREATE TABLE permisos (
    permiso_id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) UNIQUE NOT NULL,
    recurso VARCHAR(100) NOT NULL,
    accion VARCHAR(50) NOT NULL,
    descripcion TEXT
);

-- Roles-Permisos
CREATE TABLE roles_permisos (
    rol_id INTEGER REFERENCES roles(rol_id),
    permiso_id INTEGER REFERENCES permisos(permiso_id),
    PRIMARY KEY (rol_id, permiso_id)
);

-- Auditoría general
CREATE TABLE auditoria_general (
    auditoria_id BIGSERIAL PRIMARY KEY,
    usuario_id INTEGER,
    accion VARCHAR(50) NOT NULL,
    tabla_afectada VARCHAR(100),
    registro_id INTEGER,
    datos_antiguos JSONB,
    datos_nuevos JSONB,
    ip_address INET,
    user_agent TEXT,
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

7.3 Funciones y Triggers Esenciales

Generar Folio de Venta

CREATE OR REPLACE FUNCTION generar_folio_venta()
RETURNS TRIGGER AS $$
DECLARE
    v_folio VARCHAR(20);
    v_consecutivo INTEGER;
BEGIN
    -- Obtener consecutivo del día
    SELECT COALESCE(MAX(
        SUBSTRING(folio FROM 8)::INTEGER
    ), 0) + 1 INTO v_consecutivo
    FROM ventas
    WHERE folio LIKE 'V-' || TO_CHAR(CURRENT_DATE, 'YYYYMMDD') || '%';
    
    -- Generar folio: V-AAAAMMDD-NNNN
    v_folio := 'V-' || TO_CHAR(CURRENT_DATE, 'YYYYMMDD') || '-' || LPAD(v_consecutivo::TEXT, 4, '0');
    
    NEW.folio := v_folio;
    NEW.creado_en := CURRENT_TIMESTAMP;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_generar_folio_venta
BEFORE INSERT ON ventas
FOR EACH ROW EXECUTE FUNCTION generar_folio_venta();

Actualizar Stock al Vender

CREATE OR REPLACE FUNCTION fn_actualizar_stock_venta()
RETURNS TRIGGER AS $$
BEGIN
    -- Restar del stock
    UPDATE productos 
    SET stock_actual = stock_actual - NEW.cantidad
    WHERE producto_id = NEW.producto_id;
    
    -- Registrar movimiento
    INSERT INTO inventario_movimientos (producto_id, tipo, cantidad, referencia)
    VALUES (NEW.producto_id, 'salida', NEW.cantidad, 'Venta: ' || NEW.venta_id);
    
    -- Verificar stock mínimo
    IF (SELECT stock_actual FROM productos WHERE producto_id = NEW.producto_id) 
       < (SELECT stock_minimo FROM productos WHERE producto_id = NEW.producto_id) THEN
        RAISE NOTICE 'Producto % por debajo del stock mínimo', NEW.producto_id;
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_actualizar_stock_venta
AFTER INSERT ON ventas_detalle
FOR EACH ROW EXECUTE FUNCTION fn_actualizar_stock_venta();

Calcular Totales de Venta

CREATE OR REPLACE FUNCTION fn_calcular_totales_venta()
RETURNS TRIGGER AS $$
DECLARE
    v_subtotal DECIMAL;
    v_descuento DECIMAL;
    v_impuesto DECIMAL;
BEGIN
    SELECT 
        COALESCE(SUM(subtotal), 0),
        COALESCE(SUM(descuento), 0)
    INTO v_subtotal, v_descuento
    FROM ventas_detalle 
    WHERE venta_id = NEW.venta_id;
    
    v_impuesto := (v_subtotal - v_descuento) * 0.16;  -- 16% IVA
    
    UPDATE ventas 
    SET 
        subtotal = v_subtotal,
        descuento = v_descuento,
        impuesto = v_impuesto,
        total = v_subtotal - v_descuento + v_impuesto,
        actualizado_en = CURRENT_TIMESTAMP
    WHERE venta_id = NEW.venta_id;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_calcular_totales_venta
AFTER INSERT OR UPDATE ON ventas_detalle
FOR EACH ROW EXECUTE FUNCTION fn_calcular_totales_venta();

7.4 Vistas y Reportes

Vistas de Reportes

-- Ventas diarias
CREATE VIEW reporte_ventas_diarias AS
SELECT 
    DATE(creado_en) as fecha,
    COUNT(*) as num_ventas,
    SUM(subtotal) as subtotal,
    SUM(descuento) as descuentos,
    SUM(impuesto) as impuestos,
    SUM(total) as total_ventas
FROM ventas
GROUP BY DATE(creado_en)
ORDER BY fecha DESC;

-- Productos más vendidos
CREATE VIEW reporte_productos_vendidos AS
SELECT 
    p.producto_id,
    p.sku,
    p.nombre,
    c.nombre as categoria,
    SUM(vd.cantidad) as total_vendido,
    SUM(vd.subtotal) as total_ingresos,
    p.stock_actual
FROM productos p
JOIN ventas_detalle vd ON p.producto_id = vd.producto_id
JOIN ventas v ON vd.venta_id = v.venta_id
JOIN categorias_producto c ON p.categoria_id = c.categoria_id
WHERE v.estado != 'cancelada'
GROUP BY p.producto_id, p.sku, p.nombre, c.nombre, p.stock_actual
ORDER BY total_vendido DESC;

-- Empleados por departamento
CREATE VIEW reporte_empleados_departamento AS
SELECT 
    d.departamento_id,
    d.nombre as departamento,
    COUNT(e.empleado_id) as num_empleados,
    AVG(e.salario) as salario_promedio,
    SUM(e.salario) as nomina_total
FROM departamentos d
LEFT JOIN puestos p ON d.departamento_id = p.departamento_id
LEFT JOIN empleados e ON p.puesto_id = e.puesto_id AND e.activo = true
GROUP BY d.departamento_id, d.nombre;

-- Stock por debajo del mínimo
CREATE VIEW reporte_stock_bajo AS
SELECT 
    p.producto_id,
    p.sku,
    p.nombre,
    p.stock_actual,
    p.stock_minimo,
    p.stock_maximo,
    prov.nombre as proveedor,
    p.costo,
    (p.stock_minimo - p.stock_actual) as cantidad_a_reponer
FROM productos p
JOIN proveedores prov ON p.proveedor_id = prov.proveedor_id
WHERE p.stock_actual < p.stock_minimo
AND p.activo = true
ORDER BY cantidad_a_reponer DESC;

-- Clientes morosos
CREATE VIEW reporte_clientes_morosos AS
SELECT 
    c.cliente_id,
    c.nombre,
    c.email,
    c.saldo_pendiente,
    c.credito_limite,
    (c.saldo_pendiente / c.credito_limite * 100) as porcentaje_usado,
    MAX(v.creado_en) as ultima_compra
FROM clientes c
JOIN ventas v ON c.cliente_id = v.cliente_id
WHERE c.saldo_pendiente > 0
GROUP BY c.cliente_id, c.nombre, c.email, c.saldo_pendiente, c.credito_limite
ORDER BY saldo_pendiente DESC;

7.6 Consultas de Ejemplo

-- Ventas del día
SELECT * FROM ventas 
WHERE DATE(creado_en) = CURRENT_DATE;

-- Total vendido por vendedor este mes
SELECT 
    e.nombre as vendedor,
    COUNT(v.venta_id) as num_ventas,
    SUM(v.total) as total_vendido
FROM ventas v
JOIN empleados e ON v.vendedor_id = e.empleado_id
WHERE DATE_TRUNC('month', v.creado_en) = DATE_TRUNC('month', CURRENT_DATE)
GROUP BY e.empleado_id, e.nombre;

-- Productos que necesitan reorden
SELECT * FROM reporte_stock_bajo;

-- Historial de movimientos de un producto
SELECT 
    m.creado_en,
    m.tipo,
    m.cantidad,
    m.referencia,
    p.stock_actual
FROM inventario_movimientos m
JOIN productos p ON m.producto_id = p.producto_id
WHERE m.producto_id = 1
ORDER BY m.creado_en DESC;

-- Estado de cuenta de cliente
SELECT 
    v.folio,
    v.creado_en as fecha,
    v.total,
    COALESCE(p.monto, 0) as pagado,
    v.total - COALESCE(p.monto, 0) as pendiente
FROM ventas v
LEFT JOIN pagos p ON v.venta_id = p.venta_id
WHERE v.cliente_id = 1
ORDER BY v.creado_en DESC;
